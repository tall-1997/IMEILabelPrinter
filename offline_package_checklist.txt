[ ] 1. python-3.11.9-amd64.exe

    ├── README_OFFLINE.md            # 离线安装指南
    ├── config.json                  # 配置文件模板
    ├── core/
    │   ├── bartender.py             # BarTender 连接模块（v1.2.2 修复版）
    │   ├── excel_manager.py         # Excel 数据管理
    │   ├── printer.py               # 打印控制
    │   └── validator.py             # 数据校验
    ├── ui/
    │   ├── main_window.py           # 主窗口
    │   ├── print_tab.py             # 打印界面
    │   ├── history_tab.py           # 历史记录
    │   └── templates/               # UI 模板文件
    └── resources/                   # 资源文件
